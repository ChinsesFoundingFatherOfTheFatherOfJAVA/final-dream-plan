### 5.1 projectStore

```typescript
// stores/project.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import type { ProjectItem } from '@/types/project'
import * as projectApi from '@/api/project'

export const useProjectStore = defineStore('project', () => {
  const projects = ref<ProjectItem[]>([])
  const currentProject = ref<ProjectItem | null>(null)
  const loading = ref(false)
  const pagination = ref({ page: 1, pageSize: 20, total: 0 })
  const filters = ref({
    status: '',
    type: '',
    keyword: '',
    responsible: ''
  })

  async function fetchProjects() {
    loading.value = true
    try {
      const res = await projectApi.getProjectList({
        page: pagination.value.page,
        pageSize: pagination.value.pageSize,
        ...filters.value
      })
      projects.value = res.data.records
      pagination.value.total = res.data.total
    } finally {
      loading.value = false
    }
  }

  async function fetchProject(id: string) {
    const res = await projectApi.getProjectDetail(id)
    currentProject.value = res.data
    return res.data
  }

  async function createProject(data: Partial<ProjectItem>) {
    const res = await projectApi.createProject(data)
    return res.data
  }

  async function updateProject(id: string, data: Partial<ProjectItem>) {
    await projectApi.updateProject(id, data)
  }

  async function publishProject(id: string) {
    await projectApi.publishProject(id)
  }

  async function deleteProject(id: string) {
    await projectApi.deleteProject(id)
  }

  return {
    projects, currentProject, loading, pagination, filters,
    fetchProjects, fetchProject, createProject, updateProject,
    publishProject, deleteProject
  }
})
```

### 5.2 flowStore

```typescript
// stores/flow.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { FlowNode, FlowEdge } from '@/types/flow'
import * as flowApi from '@/api/flow'

export const useFlowStore = defineStore('flow', () => {
  const nodes = ref<FlowNode[]>([])
  const edges = ref<FlowEdge[]>([])
  const selectedNode = ref<FlowNode | null>(null)
  const selectedEdge = ref<FlowEdge | null>(null)
  const isDirty = ref(false)
  const loading = ref(false)

  // 检查是否有开始和结束节点
  const hasStartNode = computed(() => nodes.value.some(n => n.type === 'start'))
  const hasEndNode = computed(() => nodes.value.some(n => n.type === 'end'))

  // 检测环
  function hasCycle(): boolean {
    const adjacency = new Map<string, string[]>()
    for (const edge of edges.value) {
      if (!adjacency.has(edge.source)) adjacency.set(edge.source, [])
      adjacency.get(edge.source)!.push(edge.target)
    }

    const visited = new Set<string>()
    const inStack = new Set<string>()

    function dfs(nodeId: string): boolean {
      visited.add(nodeId)
      inStack.add(nodeId)
      for (const neighbor of (adjacency.get(nodeId) || [])) {
        if (inStack.has(neighbor)) return true
        if (!visited.has(neighbor) && dfs(neighbor)) return true
      }
      inStack.delete(nodeId)
      return false
    }

    for (const node of nodes.value) {
      if (!visited.has(node.id) && dfs(node.id)) return true
    }
    return false
  }

  async function fetchFlow(projectId: string) {
    loading.value = true
    try {
      const res = await flowApi.getFlow(projectId)
      nodes.value = res.data.nodes || []
      edges.value = res.data.edges || []
    } finally {
      loading.value = false
    }
  }

  async function saveFlow(projectId: string) {
    if (hasCycle()) {
      throw new Error('流程图中存在环，请检查连线')
    }
    await flowApi.saveFlow(projectId, {
      nodes: nodes.value,
      edges: edges.value
    })
    isDirty.value = false
  }

  function addNode(node: FlowNode) {
    nodes.value.push(node)
    isDirty.value = true
  }

  function removeNode(nodeId: string) {
    nodes.value = nodes.value.filter(n => n.id !== nodeId)
    edges.value = edges.value.filter(e => e.source !== nodeId && e.target !== nodeId)
    isDirty.value = true
  }

  function addEdge(edge: FlowEdge) {
    edges.value.push(edge)
    isDirty.value = true
  }

  function updateEdge(edgeId: string, data: Partial<FlowEdge>) {
    const edge = edges.value.find(e => e.id === edgeId)
    if (edge) Object.assign(edge, data)
    isDirty.value = true
  }

  return {
    nodes, edges, selectedNode, selectedEdge, isDirty, loading,
    hasStartNode, hasEndNode, hasCycle,
    fetchFlow, saveFlow, addNode, removeNode, addEdge, updateEdge
  }
})
```

### 6.1 axios配置

```typescript
// api/request.ts
import axios from 'axios'
import { ElMessage } from 'element-plus'

const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',
  timeout: 15000,
  headers: { 'Content-Type': 'application/json' }
})

request.interceptors.response.use(
  response => {
    const { code, message, data } = response.data
    if (code === 0) return data
    ElMessage.error(message || '请求失败')
    return Promise.reject(new Error(message))
  },
  error => {
    ElMessage.error(error.message || '网络错误')
    return Promise.reject(error)
  }
)

export default request
```

### 7.3 AI生成话术弹窗 (AIGenerateDialog.vue)

```vue
<template>
  <el-dialog v-model="visible" title="AI生成话术" width="500px">
    <el-form :model="form" label-width="80px">
      <el-form-item label="对话场景" required>
        <el-input
          v-model="form.description"
          type="textarea"
          :rows="4"
          placeholder="请输入对话场景描述，例如：保险销售电话开场白"
        />
      </el-form-item>
      <el-form-item label="对话轮次" required>
        <el-input-number v-model="form.rounds" :min="1" :max="20" />
      </el-form-item>
    </el-form>
    <template #footer>
      <el-button @click="visible = false">取消</el-button>
      <el-button type="primary" @click="handleGenerate" :loading="loading">
        生成
      </el-button>
    </template>
  </el-dialog>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { generateScript } from '@/api/dify'
import { useFlowStore } from '@/stores/flow'

const visible = ref(false)
const loading = ref(false)
const form = ref({ description: '', rounds: 5 })
const flowStore = useFlowStore()

async function handleGenerate() {
  loading.value = true
  try {
    const result = await generateScript({
      description: `${form.value.description}，共${form.value.rounds}轮对话`
    })
    // 解析生成的节点并添加到画布
    // 生成的节点只有student_ask类型，形成一条链
    flowStore.nodes = result.nodes
    flowStore.edges = result.edges
    visible.value = false
  } finally {
    loading.value = false
  }
}
</script>
```
